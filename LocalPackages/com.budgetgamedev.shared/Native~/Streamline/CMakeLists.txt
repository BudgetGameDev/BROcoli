cmake_minimum_required(VERSION 3.24)
project(BudgetGameDevStreamline LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Streamline requires a Windows x64 toolchain.")
endif()
set(STREAMLINE_SDK "" CACHE PATH "Extracted NVIDIA Streamline 2.12.0 SDK")
set(UNITY_PLUGIN_API "" CACHE PATH "Unity 6000.5.10f1 Editor PluginAPI directory")
if(NOT EXISTS "${STREAMLINE_SDK}/include/sl.h" OR NOT EXISTS "${UNITY_PLUGIN_API}/IUnityGraphicsD3D12.h")
    message(FATAL_ERROR "Set STREAMLINE_SDK and UNITY_PLUGIN_API to the SDK/header directories.")
endif()
add_library(GfxPluginBudgetGameDevStreamline SHARED Bootstrap.cpp Imports.cpp Present.cpp Frames.cpp Pcl.cpp Diagnostics.cpp)
target_compile_features(GfxPluginBudgetGameDevStreamline PRIVATE cxx_std_17)
target_include_directories(GfxPluginBudgetGameDevStreamline SYSTEM PRIVATE "${STREAMLINE_SDK}/include" "${UNITY_PLUGIN_API}")
target_link_libraries(GfxPluginBudgetGameDevStreamline PRIVATE d3d12 dxgi dxguid wintrust crypt32 shell32 ole32)
set_target_properties(GfxPluginBudgetGameDevStreamline PROPERTIES PREFIX "")
if(MSVC)
    target_compile_options(GfxPluginBudgetGameDevStreamline PRIVATE /W4 /EHsc)
    set_property(TARGET GfxPluginBudgetGameDevStreamline PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_link_options(GfxPluginBudgetGameDevStreamline PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
install(TARGETS GfxPluginBudgetGameDevStreamline RUNTIME DESTINATION .)
foreach(file sl.interposer.dll sl.common.dll sl.dlss_g.dll sl.reflex.dll sl.pcl.dll nvngx_dlssg.dll nvngx_dlss.dll nvngx_dlss.license.txt reflex.license.txt)
    install(FILES "${STREAMLINE_SDK}/bin/x64/${file}" DESTINATION .)
endforeach()
install(FILES "${STREAMLINE_SDK}/license.txt" DESTINATION . RENAME streamline.license.txt)
